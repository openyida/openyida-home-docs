```javascript
#!/usr/bin/env node
import { program } from 'commander';
import { createApp } from '../lib/app.js';

program
  .name('openyida')
  .description('AI-powered Yida CLI')
  .version('1.0.0');

program
  .command('create-app <name>')
  .description('Create a new Yida app')
  .action(createApp);

program.parse();
```

```
User Input
    ↓
CLI Parser (commander)
    ↓
Parameter Validation
    ↓
Auth Check
    ↓
Business Logic
    ↓
Yida API Call
    ↓
Result Processing
    ↓
Output Display
```

```
Error Occurs
    ↓
Catch at Boundary
    ↓
Error Classification
    ↓
User-Friendly Message
    ↓
Suggested Solution
    ↓
Exit with Code
```
