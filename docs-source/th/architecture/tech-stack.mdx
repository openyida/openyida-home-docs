# สแต็กเทคโนโลยี

หน้านี้อธิบาย สแต็กเทคโนโลยี ของ OpenYida โดยสรุปเป้าหมาย ขั้นตอนการใช้งาน และข้อควรระวังที่เกี่ยวข้องสำหรับการสร้างแอป Yida ด้วย AI

## ภาพรวม

เอกสารนี้อธิบาย สแต็กเทคโนโลยี ใน OpenYida พร้อมแนวทางการใช้งาน คำสั่ง และข้อควรทราบที่เกี่ยวข้อง

## ประเด็นสำคัญ

- ใช้หน้านี้เพื่อทำความเข้าใจบริบทหลักก่อนเริ่มลงมือในโครงการจริง
- ชื่อคำสั่ง พารามิเตอร์ โค้ด และชื่อผลิตภัณฑ์บางส่วนคงไว้เป็นภาษาอังกฤษเพื่อให้ค้นหาและคัดลอกได้ถูกต้อง
- เมื่อใช้งานกับ宜搭จริง ให้ตรวจสอบสิทธิ์ องค์กร และสภาพแวดล้อมก่อนเผยแพร่

## การใช้งาน

ปฏิบัติตามขั้นตอนในเอกสารนี้ร่วมกับคำสั่ง OpenYida CLI และการตั้งค่าขององค์กร หากมีตัวอย่างโค้ดด้านล่าง สามารถคัดลอกไปปรับใช้ได้โดยตรง

## คำสั่งและตัวอย่าง

```javascript
#!/usr/bin/env node
import { program } from 'commander';
import { createApp } from '../lib/app.js';

program
  .name('openyida')
  .description('AI-powered Yida CLI')
  .version('1.0.0');

program
  .command('create-app <name>')
  .description('Create a new Yida app')
  .action(createApp);

program.parse();
```

User Input
    ↓

    ↓

    ↓

    ↓

    ↓

    ↓

async function createApp(name: string): Promise<void> {
  const config = await loadConfig();
  await validateName(name);
  await api.createApp({ name, config });
}
